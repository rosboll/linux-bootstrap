# Base packages — installed on every profile (desktop and server).
# Lines starting with # are comments. Empty lines are ignored.
# Trailing inline comments are stripped before installation.

# Locale support. 20-locale.sh edits /etc/locale.gen and runs locale-gen;
# both come from this package. Debian's standard-system task usually pulls
# it in, but minimal Ubuntu cloud images do not — without it 20-locale.sh
# degrades to warnings and sets LANG to a locale that was never generated.
locales

# Network CLI (no GUI)
# Note: 'nc' from netcat-openbsd is preferred over Trixie's default
# netcat-traditional — better IPv6 support, -N (shutdown on EOF), and it's
# the variant nearly all tooling and write-ups assume. Note that Debian
# builds netcat-openbsd WITHOUT -e; use ncat (in this list) if you need it.
# 10-packages.sh pins update-alternatives accordingly.
nmap
ncat
netcat-openbsd
tcpdump
whois
mtr-tiny
traceroute

# Containers — useful on both workstation and server profile.
#
# Docker comes from Docker's own apt repo (download.docker.com), not from
# the distro's docker.io. 10-packages.sh configures that repo and removes
# any conflicting distro Docker packages first — see DISTRO_DOCKER_PKGS in
# common.sh for why the two stacks can't coexist.
#
# This is the upstream-recommended package set: engine, CLI, containerd,
# and the buildx/compose v2 CLI plugins ('docker compose', no hyphen).
docker-ce
docker-ce-cli
containerd.io
docker-buildx-plugin
docker-compose-plugin

# Inbound SSH access. 45-ssh-authorized-keys.sh runs `ssh-import-id
# gh:rosboll` to pull the account's published keys into
# ~/.ssh/authorized_keys. Both profiles: on servers it satisfies
# 70-ssh-hardening.sh's lock-out precheck, on workstations it's what lets
# the box accept SSH from the rest of the fleet. Preinstalled on Ubuntu
# cloud images, not on Debian.
ssh-import-id

# GPG (base — YubiKey-specific packages live in desktop.txt)
gnupg
